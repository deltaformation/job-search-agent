  createdAt: text("created_at").notNull(), updatedAt: text("updated_at").notNull(),

  foreignKey({columns:[t.parentVersionId,t.jobId],foreignColumns:[resumeVersions.id,resumeVersions.jobId]}),

  c("v13_ws_revision","revision>=1"),
